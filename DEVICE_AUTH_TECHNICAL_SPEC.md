**Purpose:** Create anonymous device profile without email/wallet

**Request:**

**Response (Success):**
```typescript
interface DeviceRegisterResponse {
  success: true

    authType: 'device'
```

**Response (Error):**

**Implementation:**
```typescript
// app/api/auth/device/register/route.ts
import { NextRequest, NextResponse } from 'next/server'
import { prisma } from '@/lib/prisma'
import jwt from 'jsonwebtoken'
import crypto from 'crypto'

export async function POST(req: NextRequest) {
  try {
    const { deviceId, fingerprint } = await req.json()
    
    // Validate device ID (UUID format)
    if (!isValidUUID(deviceId)) {
      return NextResponse.json(
        { error: 'Invalid device ID format', code: 'INVALID_DEVICE_ID' },
        { status: 400 }
      )
    }
    
    // Check if device already registered
    const existingUser = await prisma.user.findFirst({
      where: { deviceId }
    })
    
    if (existingUser) {
      // Return existing user's token
      const token = jwt.sign(
        { userId: existingUser.id, deviceId, authType: 'device' },
        process.env.NEXTAUTH_SECRET!,
        { expiresIn: '30d' }
      )
      
      // Update last seen
      await prisma.user.update({
        where: { id: existingUser.id },
        data: { deviceLastSeen: new Date() }
      })
      
      return NextResponse.json({
        success: true,
        user: {
          id: existingUser.id,
          username: existingUser.nickname,
          authType: 'device',
          deviceId: existingUser.deviceId,
          createdAt: existingUser.createdAt
        },
        token,
        expiresIn: 30 * 24 * 60 * 60 // 30 days in seconds
      })
    }
    
    // Create new anonymous user
    const username = `anon_${deviceId.slice(0, 8)}`
    
    // Hash fingerprint for backup identification
    const fingerprintHash = fingerprint
      ? crypto.createHash('sha256').update(JSON.stringify(fingerprint)).digest('hex')
      : null
    
    const user = await prisma.user.create({
      data: {
        deviceId,
        deviceFingerprint: fingerprintHash,
        deviceCreatedAt: new Date(),
        deviceLastSeen: new Date(),
        nickname: username,
        authType: 'device',
        isCreator: true // Can post free content
      }
    })
    
    // Create device binding record
    await prisma.deviceBinding.create({
      data: {
        userId: user.id,
        deviceId,
        deviceFingerprint: fingerprintHash,
        userAgent: fingerprint?.userAgent,
        ipAddress: req.headers.get('x-forwarded-for') || req.headers.get('x-real-ip'),
        createdAt: new Date(),
        lastSeenAt: new Date(),
        isActive: true
      }
    })
    
    // Generate JWT
    const token = jwt.sign(
      { userId: user.id, deviceId, authType: 'device' },
      process.env.NEXTAUTH_SECRET!,
      { expiresIn: '30d' }
    )
    
    return NextResponse.json({
      success: true,
      user: {
        id: user.id,
        username: user.nickname,
        authType: 'device',
        deviceId: user.deviceId,
        createdAt: user.createdAt
      },
      token,
      expiresIn: 30 * 24 * 60 * 60
    })
    
  } catch (error) {
    console.error('[Device Register] Error:', error)
    return NextResponse.json(
      { error: 'Registration failed', code: 'INTERNAL_ERROR' },
      { status: 500 }
    )
  }
}

function isValidUUID(uuid: string): boolean {
  const regex = /^[0-9a-f]{8}-[0-9a-f]{4}-4[0-9a-f]{3}-[89ab][0-9a-f]{3}-[0-9a-f]{12}$/i
  return regex.test(uuid)
}
```

```typescript
// app/api/auth/device/link/route.ts
export async function POST(req: NextRequest) {
  const { deviceId, linkMethod, ...linkData } = await req.json()
  
  // Find device user
  const user = await prisma.user.findFirst({
    where: { deviceId, authType: 'device' }
  })
  
  if (!user) {
    return NextResponse.json(
      { error: 'Device profile not found' },
      { status: 404 }
    )
  }
  
  if (linkMethod === 'wallet') {
    // Verify wallet signature
    const { wallet, signature, message } = linkData
    const isValid = await verifyWalletSignature(wallet, signature, message)
    
    if (!isValid) {
      return NextResponse.json(
        { error: 'Invalid wallet signature' },
        { status: 401 }
      )
    }
    
    // Check wallet not already used
    const existingWallet = await prisma.user.findFirst({
      where: { wallet, id: { not: user.id } }
    })
    
    if (existingWallet) {
      return NextResponse.json(
        { error: 'Wallet already linked to another account' },
        { status: 409 }
      )
    }
    
    // Upgrade user
    const updatedUser = await prisma.user.update({
      where: { id: user.id },
      data: {
        wallet,
        solanaWallet: wallet,
        authType: 'wallet'
      }
    })
    
    // Generate new token
    const token = jwt.sign(
      { userId: updatedUser.id, wallet, authType: 'wallet', deviceId },
      process.env.NEXTAUTH_SECRET!,
      { expiresIn: '30d' }
    )
    
    return NextResponse.json({
      success: true,
      user: {
        id: updatedUser.id,
        username: updatedUser.nickname,
        authType: 'wallet',
        wallet: updatedUser.wallet,
        deviceId: updatedUser.deviceId
      },
      token
    })
  }
  
  if (linkMethod === 'email') {
    // Similar implementation for email linking
    // ... (see Soft Auth implementation)
  }
  
  return NextResponse.json(
    { error: 'Invalid link method' },
    { status: 400 }
  )
```

```typescript
// app/api/auth/device/recover/route.ts
import crypto from 'crypto'

export async function POST(req: NextRequest) {
  const { fingerprint } = await req.json()
  
  // Hash fingerprint
  const fingerprintHash = crypto
    .createHash('sha256')
    .update(JSON.stringify(fingerprint))
    .digest('hex')
  
  // Find matching device bindings
  const matches = await prisma.deviceBinding.findMany({
    where: {
      deviceFingerprint: fingerprintHash,
      isActive: true
    },
    include: {
      user: {
        select: {
          id: true,
          nickname: true,
          deviceId: true,
          deviceLastSeen: true,
          postsCount: true,
          _count: {
            select: { likes: true }
          }
        }
      }
    },
    orderBy: {
      lastSeenAt: 'desc'
    },
    take: 5  // Max 5 matches
  })
  
  if (matches.length === 0) {
    return NextResponse.json({
      success: false,
      error: 'No matching device profiles found'
    }, { status: 404 })
  }
  
  // Calculate confidence scores
  const results = matches.map(match => {
    const daysSinceLastSeen = Math.floor(
      (Date.now() - match.lastSeenAt.getTime()) / (1000 * 60 * 60 * 24)
    )
    
    // Confidence score based on recency
    const confidenceScore = Math.max(0, 100 - daysSinceLastSeen * 5)
    
    return {
      deviceId: match.deviceId,
      username: match.user.nickname!,
      lastSeen: match.lastSeenAt.toISOString(),
      postsCount: match.user.postsCount,
      likesCount: match.user._count.likes,
      confidenceScore
    }
  })
  
  return NextResponse.json({
    success: true,
    matches: results
  })
}
```

## 🎨 Frontend Implementation

### Device ID Generation

```typescript
// lib/auth/device.ts
import { v4 as uuidv4 } from 'uuid'

export class DeviceAuth {
  private static readonly STORAGE_KEY = 'fonana_device_id'
  private static readonly FINGERPRINT_KEY = 'fonana_device_fingerprint'
  
  /**
   * Get or create device ID
   */
  static getDeviceId(): string {
    // Check localStorage first
    let deviceId = localStorage.getItem(this.STORAGE_KEY)
    
    if (!deviceId) {
      // Generate new UUID
      deviceId = uuidv4()
      localStorage.setItem(this.STORAGE_KEY, deviceId)
      console.log('[DeviceAuth] New device ID created:', deviceId)
    }
    
    return deviceId
  }
  
  /**
   * Generate light fingerprint (non-invasive)
   */
  static generateFingerprint(): {
    userAgent: string
    screenResolution: string
    timezone: string
    language: string
    platform: string
  } {
    return {
      userAgent: navigator.userAgent,
      screenResolution: `${screen.width}x${screen.height}`,
      timezone: Intl.DateTimeFormat().resolvedOptions().timeZone,
      language: navigator.language,
      platform: navigator.platform
    }
  }
  
  /**
   * Register device with backend
   */
  static async register(): Promise<{
    success: boolean
    user?: any
    token?: string
  }> {
    const deviceId = this.getDeviceId()
    const fingerprint = this.generateFingerprint()
    
    try {
      const res = await fetch('/api/auth/device/register', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ deviceId, fingerprint })
      })
      
      const data = await res.json()
      
      if (data.success) {
        // Save token
        localStorage.setItem('fonana_token', data.token)
        
        // Save fingerprint for recovery
        localStorage.setItem(
          this.FINGERPRINT_KEY,
          JSON.stringify(fingerprint)
        )
      }
      
      return data
    } catch (error) {
      console.error('[DeviceAuth] Registration failed:', error)
      return { success: false }
    }
  }
  
  /**
   * Check if device is registered
   */
  static isRegistered(): boolean {
    return !!localStorage.getItem(this.STORAGE_KEY) &&
           !!localStorage.getItem('fonana_token')
  }
  
  /**
   * Clear device data (logout)
   */
  static clear(): void {
    localStorage.removeItem(this.STORAGE_KEY)
    localStorage.removeItem(this.FINGERPRINT_KEY)
    localStorage.removeItem('fonana_token')
  }
}
```

### Auto-Registration Component

```typescript
// components/DeviceAuthProvider.tsx
'use client'

import { useEffect, useState } from 'react'
import { DeviceAuth } from '@/lib/auth/device'
import { useUser } from '@/lib/store/appStore'
import { useRouter } from 'next/navigation'

export function DeviceAuthProvider({ children }: { children: React.ReactNode }) {
  const user = useUser()
  const router = useRouter()
  const [isRegistering, setIsRegistering] = useState(false)

  useEffect(() => {
    // Skip if user already authenticated
    if (user) return
    
    // Skip if already registering
    if (isRegistering) return
    
    // Auto-register device
    async function autoRegister() {
      setIsRegistering(true)
      
      console.log('[DeviceAuth] Auto-registering device...')
      
      const result = await DeviceAuth.register()
      
      if (result.success) {
        console.log('[DeviceAuth] Device registered:', result.user)
        
        // Fetch user data
        const res = await fetch('/api/user', {
          headers: {
            'Authorization': `Bearer ${result.token}`
          }
        })
        
        if (res.ok) {
          const userData = await res.json()
          // Update app store (trigger re-render)
          window.location.reload()
        }
      } else {
        console.error('[DeviceAuth] Registration failed')
      }
      
      setIsRegistering(false)
    }
    
    // Delay to avoid blocking initial render
    const timer = setTimeout(autoRegister, 500)
    
    return () => clearTimeout(timer)
  }, [user, isRegistering, router])

  return <>{children}</>
}
```

### Device Binding Cleanup (Cron Job)

```typescript
// scripts/cleanup-device-bindings.ts
import { prisma } from '@/lib/prisma'

/**
 * Cleanup inactive device bindings (not seen for 90 days)
 * Run this daily via cron job
 */
export async function cleanupDeviceBindings() {
  const ninetyDaysAgo = new Date(Date.now() - 90 * 24 * 60 * 60 * 1000)
  
  // Mark old bindings as inactive
  const result = await prisma.deviceBinding.updateMany({
    where: {
      lastSeenAt: {
        lt: ninetyDaysAgo
      },
      isActive: true
    },
    data: {
      isActive: false
    }
  })
  
  console.log(`[Cleanup] Marked ${result.count} device bindings as inactive`)
  
  // Delete users with only inactive device bindings (no other auth method)
  const usersToDelete = await prisma.user.findMany({
    where: {
      authType: 'device',
      deviceLastSeen: {
        lt: ninetyDaysAgo
      },
      email: null,
      wallet: null,
      telegramId: null
    },
    select: { id: true }
  })
  
  if (usersToDelete.length > 0) {
    await prisma.user.deleteMany({
      where: {
        id: {
          in: usersToDelete.map(u => u.id)
        }
      }
    })
    
    console.log(`[Cleanup] Deleted ${usersToDelete.length} inactive device-only users`)
  }
}
```

```sql
-- Device auth adoption rate
SELECT 
  COUNT(*) FILTER (WHERE "authType" = 'device') AS device_users,
  COUNT(*) AS total_users,
  ROUND(100.0 * COUNT(*) FILTER (WHERE "authType" = 'device') / COUNT(*), 2) AS adoption_rate
FROM users
WHERE "createdAt" >= NOW() - INTERVAL '30 days';

-- Device persistence rate (D7)
SELECT 
  COUNT(*) AS total_device_users,
  COUNT(*) FILTER (WHERE "deviceLastSeen" >= NOW() - INTERVAL '7 days') AS active_d7,
  ROUND(100.0 * COUNT(*) FILTER (WHERE "deviceLastSeen" >= NOW() - INTERVAL '7 days') / COUNT(*), 2) AS persistence_rate
FROM users
WHERE "authType" = 'device'
  AND "createdAt" >= NOW() - INTERVAL '30 days'
  AND "createdAt" <= NOW() - INTERVAL '7 days';

-- Device → Full upgrade rate
SELECT 
  COUNT(*) AS total_device_users,
  COUNT(*) FILTER (WHERE "authType" IN ('wallet', 'soft', 'telegram') AND "deviceId" IS NOT NULL) AS upgraded_users,
  ROUND(100.0 * COUNT(*) FILTER (WHERE "authType" IN ('wallet', 'soft', 'telegram') AND "deviceId" IS NOT NULL) / COUNT(*), 2) AS upgrade_rate
FROM users
WHERE "deviceCreatedAt" IS NOT NULL
  AND "deviceCreatedAt" >= NOW() - INTERVAL '30 days';
```
